デップマップ図を一括作成する方法

1.Download

  1. https://depmap.org/portal/data_page/?tab=currentRelease

     

    ↑Click here to download the data set you want.

    例えば：CRISPRGeneEffect.csv

  2. And then, download A single gene data

    例えば：https://depmap.org/portal/gene/FKBP5?tab=dependency　

    The FKBP5 data.

    Note

    This step is only for matching the lineage, any gene is OKと思います。 Because the CRISPRGeneEffect.csv don't have lineage data.

2. R analyze

  1. Put all these files into the Working Directory , we will use RStudio to analyze.

  2. ​x
    # 必要なライブラリの読み込み
    library(data.table)
    library(dplyr)
    library(ggplot2)
    library(showtext)
    

    # Arialフォントの読み込みと有効化（PDF保存用）
    font_add("Arial", regular = "Arial.ttf")
    showtext_auto()
    

    # DepMapのCRISPRデータ（全遺伝子）を読み込む
    crispr_df <- fread("CRISPRGeneEffect.csv")
    

    # FKBPで始まる遺伝子名を含む列を抽出（例：FKBP5 (2289)）
    fkbp_cols <- grep("^FKBP", names(crispr_df), value = TRUE)
    

    # 対象の列（V1：細胞株ID + FKBP遺伝子スコア）を抽出
    data_fkbp <- crispr_df[, c("V1", fkbp_cols), with = FALSE]
    

    # ---- Lineage（がん種分類）情報の統合 ----
    # FKBP5の個別ファイルに含まれる細胞株情報を利用
    meta <- fread("FKBP5 CRISPR (DepMap Public 24Q4+Score Chronos).csv")
    

    # "Depmap ID" 列を "V1" としてリネームし、Lineage列と一緒に抽出
    lineage_map <- meta[, .(V1 = `Depmap ID`, Lineage)]
    

    # CRISPRスコアのデータとLineage情報をDepMap ID（V1）でマージ
    data_merged <- merge(data_fkbp, lineage_map, by = "V1", all.x = TRUE)
    

    # プロットの保存用フォルダを作成（既に存在する場合は警告のみ表示）
    dir.create("FKBP_CRISPR_Plots", showWarnings = FALSE)
    

    # --- FKBP遺伝子ごとにループしてボックスプロットを作成 ---
    for (gene in fkbp_cols) {
      
      # 指定したFKBP遺伝子のスコアとLineage情報を抽出
      df <- data_merged %>%
        select(V1, Lineage, all_of(gene)) %>%             # V1: 細胞ID, Lineage: がん種, gene: CRISPRスコア
        filter(!is.na(Lineage)) %>%                       # Lineageが欠損していないデータのみ使用
        group_by(Lineage) %>%
        filter(n() > 5) %>%                               # がん種ごとのサンプル数が5以上のものを抽出
        ungroup()
      
      # がん種ごとに平均スコアを計算し、スコアの符号に応じて色分け（平均 < 0：赤、≥ 0：黒）
      df <- df %>%
        group_by(Lineage) %>%
        mutate(mean_value = mean(.data[[gene]], na.rm = TRUE),
               color_group = ifelse(mean_value < 0, "red", "black")) %>%
        ungroup()
      
      # がん種の表示順をスコア順に調整（逆順でy軸に表示）
      df$Lineage <- factor(df$Lineage, levels = rev(sort(unique(as.character(df$Lineage)))))
      
      # ボックスプロットを作成
      plot <- ggplot(df, aes(x = .data[[gene]], y = Lineage)) +
        geom_vline(xintercept = 0, linetype = "solid", color = "black", linewidth = 0.5) + # x=0の線
        geom_boxplot(aes(color = color_group), outlier.shape = NA) +                      # 色分けされた箱ひげ図
        geom_point(shape = 21, color = "black", fill = "white", size = 1, stroke = 0.5) + # 各点のプロット
        labs(
          title = paste("DepMap CRISPR score:", gene),    # グラフタイトル
          x = "Gene Effect (Chronos)",                    # x軸ラベル
          y = "Tumor lineage"                             # y軸ラベル
        ) +
        theme_minimal() +
        theme(
          text = element_text(color = "black", size = 12, family = "Arial"),
          axis.title = element_text(color = "black", size = 12, family = "Arial"),
          axis.text = element_text(color = "black", size = 12, family = "Arial"),
          panel.grid.major = element_blank(),
          panel.grid.minor = element_blank(),
          panel.border = element_blank(),
          axis.line.x = element_line(color = "black", linewidth = 0.5),
          axis.line.y = element_blank()
        ) +
        scale_x_continuous(breaks = c(-1, -0.5, 0, 0.5), limits = c(-0.5, 0.5)) +
        scale_color_identity()  # 色を固定（赤・黒）
    

      # PDFとして保存
      ggsave(
        filename = paste0("FKBP_CRISPR_Plots/DepMap_", gene, "_CRISPR.pdf"), # ファイル名
        plot = plot, device = "pdf",
        dpi = 150, width = 6, height = 6,
        bg = "white"
      )
    }
